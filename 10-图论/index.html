10.01

建图与 BFS：无权最短路

知识与适用场景

邻接表用 vector<vector<int>> 表示出边，空间 O(V+E)。无向边要添加两个方向。BFS 按距离层扩展，第一次发现一个节点时已经得到最少边数。

1.1 建图前先问：状态是否包含足够的信息？

同一个格子如果带着不同钥匙、剩余消除障碍次数不同，未来可走的路也不同，便不是同一个搜索状态。此时 visited 应包含 (位置,钥匙集合) 或 (位置,剩余次数)，而不能只按格子去重。常见状态数上界是位置数乘附加状态数，复杂度也应在这个展开后的状态图上计算。

二维网格可以不显式建边：遍历上下左右，检查坐标、障碍和访问标记即可。邻接矩阵判一条边是否存在是 O(1)，但扫描一个点的所有邻居需要 O(V)；邻接表扫描是 O(出度)，判边则不天然是 O(1)。有孤立点时仍要保留它对应的空邻接表。

1.3 BFS 的层序为什么等于最少边数？

初始队列只有距离 0 的源点。FIFO 保证距离 d 的点先于距离 d+1 的点扩展；因此第一次发现 v 时，如果存在更短路径，它的倒数第二个点应更早出队并发现 v，与首次发现矛盾。这个证明只使用边数，不比较任意权重之和。

发现 v 时就设置 dist[v] 并入队，同时设置 parent[v]=u。若等到出队才标记，菱形图中的两个父节点都可能把同一个 v 加入队列。只求一条最短路径时，首次发现目标即可停；要完整距离数组或统计同层信息时，按相应目标继续处理。

vector<int> bfsDistances(const vector<vector<int>>& g, int source) {
    int n=(int)g.size();
    if (source<0 || source>=n) throw out_of_range("source");
    vector<int> dist(n,-1);
    queue<int> q;
    dist[source]=0; q.push(source);
    while (!q.empty()) {
        int u=q.front(); q.pop();
        for (int v:g[u]) {
            if (dist[v]!=-1) continue;
            dist[v]=dist[u]+1;
            q.push(v);
        }
    }
    return dist;
}

optional<vector<int>> shortestUnweightedPath(
    const vector<vector<int>>& g, int s, int t) {
    int n = (int)g.size();
    if (s < 0 || s >= n || t < 0 || t >= n)
        throw out_of_range("vertex");
    vector<int> dist(n, -1), parent(n, -1);
    queue<int> q;
    dist[s] = 0; q.push(s);
    while (!q.empty() && dist[t] == -1) {
        int u = q.front(); q.pop();
        for (int v : g[u]) if (dist[v] == -1) {
            dist[v] = dist[u] + 1;
            parent[v] = u;
            q.push(v);
        }
    }
    if (dist[t] == -1) return nullopt;
    vector<int> path;
    for (int v = t; v != -1; v = parent[v]) path.push_back(v);
    reverse(path.begin(), path.end());
    return path;
}

1.6 多源 BFS：到最近源点的网格距离

约定 -1 是障碍、0 是源点、1 是可通行格；四方向移动代价均为 1。先把所有源点放进同一队列，不逐个源点重复搜索。返回 -1 表示障碍或不可达，可结合原网格区分。输入不被修改，无源点时所有距离保持 -1。时间、空间均 O(RC)。

可把它理解成一个虚拟源点以零代价连接全部源点，但实现直接初始化距离 0，不需要真的插入零权边后套普通 BFS。腐烂橘子题中，应把烂橘子视为源、鲜橘子视为待到达点、空格视为不可走，最后检查所有鲜橘子是否可达并取最大距离。

vector<vector<int>> nearestSources(const vector<vector<int>>& grid) {
    int r = (int)grid.size();
    if (!r) return {};
    int c = (int)grid[0].size();
    for (const auto& row : grid) {
        if ((int)row.size() != c) throw invalid_argument("ragged grid");
        for (int x : row) if (x < -1 || x > 1)
            throw invalid_argument("cell");
    }
    vector<vector<int>> d(r, vector<int>(c, -1));
    queue<pair<int,int>> q;
    for (int i = 0; i < r; ++i) for (int j = 0; j < c; ++j)
        if (grid[i][j] == 0) { d[i][j] = 0; q.push({i,j}); }
    const int di[] = {-1,1,0,0}, dj[] = {0,0,-1,1};
    while (!q.empty()) {
        auto [i,j] = q.front(); q.pop();
        for (int k = 0; k < 4; ++k) {
            int x = i + di[k], y = j + dj[k];
            if (x < 0 || x >= r || y < 0 || y >= c) continue;
            if (grid[x][y] == -1 || d[x][y] != -1) continue;
            d[x][y] = d[i][j] + 1; q.push({x,y});
        }
    }
    return d;
}

1.7 无向图判环：用边编号跳过来路

简单无向图可以跳过父节点，但多重图的两条平行边本身构成长为 2 的环，只跳过父节点会漏报。给每条输入无向边一个唯一编号，两个方向共用编号，只跳过刚走来的那条边。自环在本约定下也是环。显式栈避免深链导致的递归栈溢出。时间 O(V+E)，图和辅助存储 O(V+E)。

bool hasUndirectedCycle(int n, const vector<pair<int,int>>& edges) {
    if (n < 0) throw invalid_argument("n");
    vector<vector<pair<int,int>>> g(n);
    for (int id = 0; id < (int)edges.size(); ++id) {
        auto [u,v] = edges[id];
        if (u < 0 || u >= n || v < 0 || v >= n) throw out_of_range("edge");
        g[u].push_back({v,id}); g[v].push_back({u,id});
    }
    vector<char> seen(n, false);
    for (int s = 0; s < n; ++s) if (!seen[s]) {
        stack<pair<int,int>> st;
        seen[s] = true; st.push({s,-1});
        while (!st.empty()) {
            auto [u, incoming] = st.top(); st.pop();
            for (auto [v,id] : g[u]) {
                if (id == incoming) continue;
                if (seen[v]) return true;
                seen[v] = true; st.push({v,id});
            }
        }
    }
    return false;
}

1.8 二分图：把相邻点染成相反颜色

前提是无向图（邻接表双向存边）。每个未染色分量都启动 BFS，遇到同色相邻点立即失败。二分图等价于不存在奇环，不等价于无环：四边形有环仍是二分图。自环一定失败。时间 O(V+E)，辅助空间 O(V)。

bool isBipartiteGraph(const vector<vector<int>>& g) {
    int n = (int)g.size();
    vector<int> color(n, -1);
    queue<int> q;
    for (int s = 0; s < n; ++s) if (color[s] == -1) {
        color[s] = 0; q.push(s);
        while (!q.empty()) {
            int u = q.front(); q.pop();
            for (int v : g[u]) {
                if (color[v] == -1) {
                    color[v] = color[u] ^ 1; q.push(v);
                } else if (color[v] == color[u]) return false;
            }
        }
    }
    return true;
}

面试追问：关键点完整解答

为什么 DFS 找到的第一条路径不保证最短？

DFS 先沿一条分支走到底，可能先找到 100 条边的路径，而另一条尚未探索的分支只需 2 条边。遍历顺序没有按距离排序。DFS 可以枚举所有简单路径再取最短，但路径数可能指数级；无权最短路直接使用 BFS。

如何记录最短路径？不可达和源点等于终点怎么处理？

在首次发现 v 的同一步设置 parent[v]=u，不在以后反复覆盖。到终点后沿父指针走到源点，再反转。不可达单独返回 nullopt；s=t 返回 [s]，边数为 0。若要所有最短路径，需要保存满足 dist[v]=dist[u]+1 的全部前驱，枚举输出仍可能指数级。

多源 BFS 为什么取的是最近源，而不是所有源距离之和？

所有源都从第 0 层出发，第一次发现一个点的是最先到达它的波前，所以距离等于各源距离的最小值。若要求到所有建筑的距离和，需要分别搜索再累加，不能把全部建筑同时入队替代。

求连通分量时，有向图能直接照抄无向图外层循环吗？

无向图从任一未访问点可恰好遍历它的整个连通分量。对有向图，单向可达不是等价关系；外层遍历次数依赖起点顺序，并不等于强连通分量数。弱连通可忽略方向，强连通要求互相可达，需要 Tarjan 或 Kosaraju 等专门算法。

什么时候用全局 visited，什么时候要回溯恢复？

只求可达性或连通块时，全局 visited 防止重复访问，不恢复。枚举所有简单路径时，onPath 只表示当前路径占用，退出当前分支后应恢复；否则会丢掉经过同一节点的其他合法路径。状态带资源时先确认状态定义充分，再讨论能否合并。

为什么二分图与奇环等价？断开的图怎么检查？

相邻点颜色交替，沿奇数条边回到起点要求起点与自己颜色相反，产生矛盾。反过来，若 BFS 染色出现同色相邻点，可由搜索树路径加冲突边构成奇环。必须从每个未染色节点启动，因为某个未到达分量可能藏着奇环。

10.02

拓扑排序与有向环

知识与适用场景

拓扑序要求每条边 u→v 中 u 都出现在 v 前面。只有 DAG 存在完整拓扑序；入度为 0 的节点表示当前没有尚未完成的前置依赖。

2.1 先修关系的方向，必须先读懂

课程表常给出 [a,b] 表示学 a 前先学 b，因此应建 b→a，并给 a 的入度加一。入度的含义是当前还有多少前置任务未完成，不是总出边数。统计入度和删除出边必须使用同一份边集合：保留平行边时两边都计数，去重时也要同步去重。

Kahn 的正确性来自：DAG 的任意非空子图都有入度为 0 的点。否则一直沿前驱回退，有限节点中必然重复而构成环。每次移除一个零入度点不会违背任何尚存依赖，重复后得到合法序列。

vector<int> topoSort(const vector<vector<int>>& g) {
    int n=(int)g.size();
    vector<int> degree(n,0), order;
    for (const auto& edges:g) for (int v:edges) ++degree[v];
    queue<int> q;
    for (int u=0;u<n;++u) if (degree[u]==0) q.push(u);
    while (!q.empty()) {
        int u=q.front(); q.pop(); order.push_back(u);
        for (int v:g[u]) if (--degree[v]==0) q.push(v);
    }
    if ((int)order.size()!=n) return {}; // 非空图时，空结果表示有环
    return order;
}

2.4 字典序最小拓扑序与唯一性

TopologyResult lexicographicTopology(const vector<vector<int>>& g) {
    int n = (int)g.size();
    vector<int> indegree(n);
    for (const auto& row : g) for (int v : row) ++indegree[v];
    priority_queue<int, vector<int>, greater<int>> ready;
    for (int i = 0; i < n; ++i) if (!indegree[i]) ready.push(i);
    vector<int> order;
    bool unique = true;
    while (!ready.empty()) {
        if (ready.size() != 1) unique = false;
        int u = ready.top(); ready.pop(); order.push_back(u);
        for (int v : g[u]) if (--indegree[v] == 0) ready.push(v);
    }
    bool dag = (int)order.size() == n;
    return {dag, dag && unique, order}; // 有环时 order 只是可处理前缀
}

2.5 三色 DFS：返回一个实际有向环

用显式栈帧保存每个节点下一条待处理出边，模拟递归的进入和退出，灰色点才恰好对应当前调用链。仅用普通栈把所有邻居一起标灰并不满足这个含义。遇到 u→v 且 v 为灰色时，parent 链 v→…→u 加这条边组成一个环。返回首尾相同的节点序列，无环返回空数组。O(V+E) 时间、O(V) 辅助空间。

vector<int> findDirectedCycle(const vector<vector<int>>& g) {
    int n = (int)g.size();
    vector<int> color(n), parent(n, -1);
    for (int s = 0; s < n; ++s) if (!color[s]) {
        vector<pair<int,size_t>> frames;
        color[s] = 1; frames.push_back({s,0});
        while (!frames.empty()) {
            int u = frames.back().first;
            if (frames.back().second == g[u].size()) {
                color[u] = 2; frames.pop_back(); continue;
            }
            int v = g[u][frames.back().second++];
            if (!color[v]) {
                parent[v] = u; color[v] = 1; frames.push_back({v,0});
            } else if (color[v] == 1) {
                vector<int> cycle{u};
                while (cycle.back() != v) cycle.push_back(parent[cycle.back()]);
                reverse(cycle.begin(), cycle.end());
                cycle.push_back(v); return cycle;
            }
        }
    }
    return {};
}

2.6 最终安全节点：反图上删除出度为零的点

安全要求从该点出发的所有路径最终终止，不是存在一条路能到终点。原图出度为零的点先安全；一个点的所有后继都被确认安全后，它才安全。把边反过来，用剩余出度做 Kahn，最后留下的是能到达环的点，包括环上游。与普通 Kahn 留下环下游不要混淆。时间、总额外空间 O(V+E)，按编号扫描输出，无需额外排序。

vector<int> eventualSafeVertices(const vector<vector<int>>& g) {
    int n = (int)g.size();
    vector<vector<int>> reverseGraph(n);
    vector<int> remaining(n);
    queue<int> q;
    for (int u = 0; u < n; ++u) {
        remaining[u] = (int)g[u].size();
        if (!remaining[u]) q.push(u);
        for (int v : g[u]) reverseGraph[v].push_back(u);
    }
    while (!q.empty()) {
        int v = q.front(); q.pop();
        for (int u : reverseGraph[v]) if (--remaining[u] == 0) q.push(u);
    }
    vector<int> answer;
    for (int i = 0; i < n; ++i) if (!remaining[i]) answer.push_back(i);
    return answer;
}

面试追问：关键点完整解答

Kahn 剩下的每个点都在环上吗？

不一定。0→1、1→0、1→2 中，2 不在任何环上，却因为环中的 1 永远不能移除而留下。Kahn 用于判断存在环或获得可执行前缀；需要环的具体节点时用三色 DFS 找一个环，或用强连通分量分析。

如何得到字典序最小拓扑序，为什么仅排序邻接表不够？

用小顶堆维护所有当前零入度点，在整个可选集合中选择最小点。只排序某个点的邻居不能控制其他来源已经在队列中的点。例如初始可选 1、3，处理 1 后释放 2，普通队列可能给出 1、3、2，小顶堆会给出 1、2、3。

有多个零入度节点，就一定有多种拓扑序吗？

必须先确认整图是 DAG。对 DAG，两个当前零入度点互相不可能存在尚存依赖路径，可以交换先后并继续完成序列，因此不唯一。含环图即使前缀出现多个选择，也没有完整拓扑序，不能称为有多个拓扑序。

为什么有向图只用 visited 会误判环？

0→1、0→2、1→3、2→3 是 DAG。探索 2 时 3 可能已经访问完成，边 2→3 只是共享后继。只有指向当前调用链上的灰色点才会闭合成环；指向黑色点不表示当前路径回到了祖先。

最终安全与能到达终点有什么不同？

若 0 同时连向终点 1 和自环点 2，0 能到达终点却不安全，因为可以选择去 2 后永不终止。安全是所有后继都安全，属于全称条件；反图删出度的算法恰好检查这个条件。有限图中，存在无限路径等价于能到达某个环。

拓扑排序能直接算任务耗时吗？

拓扑序只保证依赖顺序。如果多个任务可并行且资源无限，应沿拓扑序计算 finish[v]=duration[v]+max(finish[u])，u 为 v 的前驱，最终取所有 finish 的最大值。简单累加所有任务时长表示串行执行；按 BFS 层数计时只适用于每个任务单位耗时且允许同层并行的模型。

10.03

Dijkstra：非负权最短路

知识与适用场景

非负边权保证当前未确定节点中距离最小者，其距离不会再通过更远节点改小。这个贪心结论依赖非负权，存在负边时不成立。

3.1 Dijkstra 定下的是最小有效出堆距离，不是首次入堆距离

dist[v] 是目前已知路径的最小长度，是候选上界；堆保存这些候选。假设有效弹出的最小距离 d[u] 还不是最优，最优路径上应有第一个尚未确定的点 x，其前驱已确定并已尝试松弛 x。由于剩余边非负，x 的候选距离不会大于该条到 u 的更短路径，应该先于 u 弹出，矛盾。

只有弹出的 d 等于当前 dist[u] 才是有效状态。不要在 push 时永久 visited；节点可以先经较差路径发现，再被改进。松弛必须用严格小于，零权环若用小于等于反复入堆，可能永不终止。只问一个目标时，可在它的有效状态出堆后停止，不能在首次发现它时停止。

vector<long long> dijkstra(
    const vector<vector<pair<int,int>>>& g, int source) {
    const long long INF=LLONG_MAX/4;
    if (source<0 || source>=(int)g.size()) throw out_of_range("source");
    vector<long long> dist(g.size(),INF);
    using State=pair<long long,int>;
    priority_queue<State,vector<State>,greater<State>> pq;
    dist[source]=0; pq.push({0,source});
    while (!pq.empty()) {
        auto [d,u]=pq.top(); pq.pop();
        if (d!=dist[u]) continue;
        for (auto [v,w]:g[u]) {
            if (w<0) throw invalid_argument("negative edge");
            if (d+w<dist[v]) {
                dist[v]=d+w;
                pq.push({dist[v],v});
            }
        }
    }
    return dist;
}
复杂度
允许重复状态的堆实现：一般写 O((V+E) log(E+2))，简单图中常写 O((V+E) log V)；堆最多 O(E)，距离 O(V)。
易错点与边界
本例假设有效路径长度小于 INF。首次入堆并不代表距离最终确定，不能在入堆时永久标记。不可达节点距离仍为 INF。

3.3 0-1 BFS：用双端队列维护两个相邻距离层

边权为 0 时松弛结果和当前层相同，推到队首；为 1 时推到队尾。双端队列里的候选距离按非递减顺序处理，扮演只需两种优先级的优先队列。仍然可以多次改善 dist，不能照搬普通 BFS 的首次入队锁定。本版本携带入队时距离并跳过过期状态，严格改善才入队。时间 O(V+E)，辅助存储保守计 O(V+E)。

vector<long long> zeroOneDistances(
    const vector<vector<pair<int,int>>>& g, int s) {
    const long long INF = LLONG_MAX / 4;
    int n = (int)g.size();
    if (s < 0 || s >= n) throw out_of_range("source");
    for (const auto& row : g) for (auto [v,w] : row) {
        if (v < 0 || v >= n) throw out_of_range("edge");
        if (w != 0 && w != 1) throw invalid_argument("not a 0/1 edge");
    }
    vector<long long> d(n, INF);
    deque<pair<long long,int>> q;
    d[s] = 0; q.push_front({0,s});
    while (!q.empty()) {
        auto [du,u] = q.front(); q.pop_front();
        if (du != d[u]) continue;
        for (auto [v,w] : g[u]) if (du + w < d[v]) {
            d[v] = du + w;
            if (w == 0) q.push_front({d[v],v});
            else q.push_back({d[v],v});
        }
    }
    return d;
}

3.4 最小体力路径：把路径代价从求和改成取最大值

输入已建好的非负权图；网格题可将相邻高度差作为权重（相减前转 long long 并按约束确保可放入 int）。路径代价是沿途最大边权，目标使这个最大值最小，松弛变为 max(du,w)。延长路径不会降低此代价，因此仍满足 Dijkstra 的确定性条件。返回 nullopt 表示不可达，s=t 代价为 0。时间 O((V+E)log(E+2))，辅助空间 O(V+E)。

不是所有修改代价公式的题都能照套 Dijkstra。必须重新证明路径扩展不会让已确定的最小候选变得更优；最小体力也可二分阈值后做可达性判断，无向图还可以用排序边与并查集。

optional<long long> minimumBottleneck(
    const vector<vector<pair<int,int>>>& g, int s, int t) {
    int n = (int)g.size();
    if (s < 0 || s >= n || t < 0 || t >= n) throw out_of_range("vertex");
    for (const auto& row : g) for (auto [v,w] : row) {
        if (v < 0 || v >= n) throw out_of_range("edge");
        if (w < 0) throw invalid_argument("negative weight");
    }
    const long long INF = LLONG_MAX / 4;
    vector<long long> d(n, INF);
    using State = pair<long long,int>;
    priority_queue<State,vector<State>,greater<State>> pq;
    d[s] = 0; pq.push({0,s});
    while (!pq.empty()) {
        auto [du,u] = pq.top(); pq.pop();
        if (du != d[u]) continue;
        if (u == t) return du;
        for (auto [v,w] : g[u]) {
            long long candidate = max(du, (long long)w);
            if (candidate < d[v]) { d[v] = candidate; pq.push({d[v],v}); }
        }
    }
    return nullopt;
}

面试追问：关键点完整解答

负权边没有形成负环，能否使用 Dijkstra？

不能据此使用标准 Dijkstra。0→1(2)、0→2(5)、2→1(-10) 没有环，却能让先确定的 1 从 2 改成 -5。某些不永久确定节点的实现会反复重入堆并碰巧算对，但不再有标准 Dijkstra 的正确性过程与复杂度保证；有负环时还可能不停更新。

为什么过期状态必须跳过？跳过之后还要 visited 吗？

懒删除堆会保留旧距离，d!=dist[u] 时其出边已经没有必要按旧代价扫描。反复扫描会破坏每个点仅在有效确定时扫描出边的复杂度分析。在非负权、严格改善入堆的版本中，距离相等检查已足够，不必再使用入堆 visited。

为什么 0-1 BFS 不在首次发现时锁定节点？

0→1 权 1、0→2 权 0、2→1 权 0。1 可能先得到 1，随后应该改善为 0。如果沿用普通 BFS 的 dist!=-1 就不更新，会返回错误答案。0-1 BFS 使用距离比较和松弛，只是把堆换成了有序的双端队列。

堆版 Dijkstra 为什么有时写 log E，有时写 log V？

本章允许同一节点的多条候选共存，堆大小可达 O(E)，一般界为 O((V+E)log(E+2))。简单图有 E≤V²，log E=O(log V)，所以常简写 log V。带任意多平行边的多重图不能不说明条件就用这个替换。

网络延迟时间怎样从 dist 得到答案？

先从消息源做 Dijkstra。任一节点为 INF，说明消息无法到达所有节点，应返回 -1；否则取所有距离最大值，而非求和，因为传播可同时发生。若只求指定目标距离，可在该目标有效出堆后提前结束。

如何还原带权最短路径？零权边会导致父指针死循环吗？

仅在严格改善 dist[v] 时记录 parent[v]=u，再从终点回溯。标准非负权 Dijkstra 的改善来自有效弹出的前驱，不会将已确定点通过相等距离回写父指针。若相等时也随意重写 parent，零权环可能使父指针成环；需要字典序最小路径时应单独设计平局规则，不能简单比较父节点编号。

10.04

Floyd 与负权算法的选择

知识与适用场景

多个点之间都要查询距离且节点数较少时，可用 Floyd。它逐步允许路径经过编号更大的中间节点。存在负边但无负环时仍适用；若负环影响某对点，最短路不再有限。

4.1 Bellman-Ford：从边数限制推导 V-1 轮

不存在源点可达负环时，任何最短路都能去掉非负环，得到最多 V-1 条边的简单路径。定义第 k 轮距离为最多使用 k 条边的最短路，从上一轮快照松弛所有边便得到下一轮。允许原地松弛的常见写法也能在 V-1 轮内收敛，但单轮可能沿边的扫描顺序传播多步，不能再把它精确解释为恰好一层边数限制。

第 V 轮若某条从源可达的边还能改善，说明存在可达负环。被改善的终点和它能到达的节点均没有有限最短路。某个负环若从源点完全不可达，不影响该源的距离；若源可达负环但负环不能到达目标，该目标仍可能有正常有限答案。

4.2 三种结果必须区分，不能用一个 -1 混在一起

4.3 Floyd 的状态是允许的中间点集合，不是边数

第 k 层只允许编号 0..k 的点作为内部中间点。最短路要么不经过 k，保留旧值；要么经过 k，拆成 i→k 与 k→j，所以转移是两段之和。k 必须位于最外层，才能确保一整轮都在同一个状态阶段。

把循环写成 i、j、k，仅做一轮可能漏更新。例：0→3→2→1，各边权 1。i=0、j=1 时，d[0][2] 还未更新，d[3][1] 也未更新，于是遗漏 0→1；稍后才得到 0→2，却不会再回头更新 j=1。

矩阵初始化 d[i][i]=0，直接边取 min，负自环保留负权而不被 0 覆盖。完成后 d[k][k]<0 表示 k 处检测到负闭合走法；若 i 可达 k 且 k 可达 j，则 (i,j) 无有限最短路。不受负环影响的有限值才可作为普通最短距离读取。

推导 / 不变量

外层必须枚举中间点 k，转移 d[i][j]=min(d[i][j],d[i][k]+d[k][j])。初始化对角为 0，直接边取最小权重，其余为 INF。

void floyd(vector<vector<long long>>& d) {
    const long long INF=LLONG_MAX/4;
    int n=(int)d.size();
    for (int k=0;k<n;++k)
        for (int i=0;i<n;++i) {
            if (d[i][k]==INF) continue;
            for (int j=0;j<n;++j) {
                if (d[k][j]==INF) continue;
                d[i][j]=min(d[i][j],d[i][k]+d[k][j]);
            }
        }
    // 如有 d[i][i] < 0，则存在负环。
}

4.4 Bellman-Ford：同时标记负环影响的所有目标

边表元素为 (起点,终点,权重)，与基础 Kruskal 的 (权重,u,v) 顺序不同。每轮使用快照，保证中间候选最多 n 条边；给定数值约束下绝对值至多 10^15，安全小于 INF。bad[v] 为真时不要读取 dist[v] 当最终答案。最后从可改善终点 BFS 扩散 bad。时间 O(VE+V²)，其中 V² 来自每轮复制距离数组；常用原地版本可省去此项。空间 O(V+E)。

    vector<long long> dist;
    vector<char> bad;
};
BellmanResult bellmanFordMarked(int n, const vector<tuple<int,int,int>>& edges, int s) {
    if (n < 1 || n > 1000000 || s < 0 || s >= n) throw invalid_argument("size/source");
    const long long INF = LLONG_MAX / 4;
    vector<vector<int>> g(n);
    for (auto [u,v,w] : edges) {
        if (u < 0 || u >= n || v < 0 || v >= n) throw out_of_range("edge");
        if (w < -1000000000 || w > 1000000000) throw invalid_argument("weight");
        g[u].push_back(v);
    }
    vector<long long> d(n, INF); d[s] = 0;
    for (int step = 1; step < n; ++step) {
        auto next = d;
        bool changed = false;
        for (auto [u,v,w] : edges) if (d[u] != INF && d[u] + w < next[v]) {
            next[v] = d[u] + w; changed = true;
        }
        d.swap(next);
        if (!changed) break;
    }
    vector<char> bad(n, false); queue<int> q;
    for (auto [u,v,w] : edges)
        if (d[u] != INF && d[u] + w < d[v] && !bad[v]) {
            bad[v] = true; q.push(v);
        }
    while (!q.empty()) {
        int u = q.front(); q.pop();
        for (int v : g[u]) if (!bad[v]) { bad[v] = true; q.push(v); }
    }
    return {d,bad};
}

4.5 Floyd：处理平行边、负自环及受影响的点对

这里用上一轮矩阵快照清楚表达动态规划状态。对于负环，中间数值可能极小，使用 -INF 作为计算下限并夹紧；它不是正常有限距离，仍通过 bad 矩阵表达语义。约束使任意简单路径绝对值低于 10^15，而 INF 约 2.3×10^18，所以夹紧不会误伤不受负环影响的有限最短路。两个处于 [-INF,INF) 的操作数相加不会越过 long long 范围。时间 O(V³)、空间 O(V²)，矩阵只适合较小 V。

FloydResult allPairsMarked(int n, const vector<tuple<int,int,int>>& edges) {
    if (n < 0 || n > 1000000) throw invalid_argument("n");
    for (auto [u,v,w] : edges) {
        if (u < 0 || u >= n || v < 0 || v >= n) throw out_of_range("edge");
        if (w < -1000000000 || w > 1000000000) throw invalid_argument("weight");
    }
    const long long INF = LLONG_MAX / 4;
    vector<vector<long long>> d(n, vector<long long>(n, INF));
    for (int i = 0; i < n; ++i) d[i][i] = 0;
    for (auto [u,v,w] : edges) d[u][v] = min(d[u][v], (long long)w);
    for (int k = 0; k < n; ++k) {
        auto old = d;
        for (int i = 0; i < n; ++i) if (old[i][k] != INF)
            for (int j = 0; j < n; ++j) if (old[k][j] != INF) {
                long long candidate = max(-INF, old[i][k] + old[k][j]);
                d[i][j] = min(old[i][j], candidate);
            }
    }
    vector<vector<char>> bad(n, vector<char>(n, false));
    for (int k = 0; k < n; ++k) if (d[k][k] < 0)
        for (int i = 0; i < n; ++i) if (d[i][k] != INF)
            for (int j = 0; j < n; ++j) if (d[k][j] != INF) bad[i][j] = true;
    return {d,bad};
}

4.6 DAG 最短路：即使有负边，也可在线性时间松弛

有向无环图没有任何环，因此无需非负边权保证。按拓扑序处理 u 时，所有可能通向 u 的前驱已经处理完成，其距离不会再改动。函数返回 nullopt 表示整图不是 DAG（包括源不可达部分存在环的情况）；有值时，数组内 INF 才表示某个点不可达。时间 O(V+E)，辅助空间 O(V)。数值条件同前。

optional<vector<long long>> dagDistances(
    const vector<vector<pair<int,int>>>& g, int s) {
    int n = (int)g.size();
    if (n < 1 || n > 1000000 || s < 0 || s >= n) throw invalid_argument("size/source");
    vector<int> indegree(n);
    for (const auto& row : g) for (auto [v,w] : row) {
        if (v < 0 || v >= n) throw out_of_range("edge");
        if (w < -1000000000 || w > 1000000000) throw invalid_argument("weight");
        ++indegree[v];
    }
    queue<int> q;
    for (int i = 0; i < n; ++i) if (!indegree[i]) q.push(i);
    const long long INF = LLONG_MAX / 4;
    vector<long long> d(n, INF); d[s] = 0;
    int processed = 0;
    while (!q.empty()) {
        int u = q.front(); q.pop(); ++processed;
        for (auto [v,w] : g[u]) {
            if (d[u] != INF) d[v] = min(d[v], d[u] + w);
            if (--indegree[v] == 0) q.push(v);
        }
    }
    if (processed != n) return nullopt;
    return d;
}

面试追问：关键点完整解答

Bellman-Ford 为什么再松弛一轮就能发现可达负环？

若没有可达负环，最短路可化为至多 V-1 条边的简单路径，V-1 轮后不应再改善。还能改善说明更长的走法带来了更小代价，重复节点形成的闭合部分必须包含负环。必须检查 dist[u]!=INF，否则会把与源点无关的边误算进去。

最多 K 条边的最短路为什么需要快照？

若在同一数组上更新，扫描 0→1 后立即扫描 1→2，单轮就能走两条边，违反层数含义。每轮从 old 读取、向 next 写入，才保证每轮最多增加一条边。最多 K 次中转通常对应最多 K+1 条边，先明确题意再定循环次数。

如何检测整张图是否存在负环，而非仅源点可达负环？

增加一个超级源点，以零权边连接所有点后运行 Bellman-Ford；等价地把所有点初始距离设为 0，再检查第 V 轮是否仍有更新。此时每个分量都可达。这个初始化只用于全图检测，不能再把输出当作原来某个源点的最短路。

Floyd 为什么不能把 k 放进最内层？

其状态定义是允许使用的中间点集合。i、j 是同一层里的所有状态，而 k 是阶段，必须先完整完成前一阶段，才能安全进入下一阶段。随意交换可能读取尚未计算的子路径；0→3→2→1 的链就是一轮 i、j、k 漏更新的反例。

无向负权边为什么特别危险？SPFA 能解决性能问题吗？

若无向边可反复走，一条负权边两方向往返就形成负闭合走法，可无限降低代价；这与要求简单路径的另一类问题不同。Bellman-Ford 可检测负环，但不能给出本不存在的有限最短路。SPFA 是基于队列的松弛优化，最坏仍可能 O(VE)，不能把它当成必然快速的替代方案。

什么时候选 Floyd、反图 Dijkstra 或多次 Dijkstra？

点数小且点对查询多，Floyd 预处理 O(V³) 后查询 O(1)。非负权稀疏图可从所需源点分别 Dijkstra；若只问所有点到同一终点，把有向边反向后从该终点做一次 Dijkstra。反图求的是路径方向倒置，不是把问题改成无向图。

10.05

最小生成树与最短路的区别

知识与适用场景

最小生成树要求用最小总边权连接所有节点，不要求从某个源到其他点的路径最短。Kruskal 按边权排序，选择连接不同连通块的最小边。

5.1 最小生成树优化的是购买边的总成本

最短路树从一个固定源出发，使源到每个点的路径最短；最小生成树没有固定源，要求用最小总边权把所有点连接起来。它们通常不是同一棵树。连通无向图的生成树有 V-1 条边，负权边不妨碍这个定义；自环不能连接新的点，平行边可以作为不同候选。

不连通图没有覆盖全部点的生成树，但可求每个分量的最小生成树，合称最小生成森林。应明确函数返回的是失败还是森林：基础 Kruskal 返回 nullopt；后面的 Prim 返回 forest 的总成本与分量数，调用者自行判断是否连通。

5.3 割性质：贪心选择为何安全？

把点集划成 S 和其补集 V−S，一条边两端分居两侧就叫跨割边。若这个割不切断此前已选择的边，取权重最小的跨割边 e，可以让已有选择扩展成某棵 MST。证明：若一棵包含已有边的 MST 没有 e，加 e 后形成环，环上必有另一条跨割边 f；w(e)≤w(f)，用 e 换掉 f 不会更差。

注意“可以属于某棵 MST”不等于“属于所有 MST”。当最小边有并列时，可能存在多种答案。Kruskal 用并查集的连通块诱导割；Prim 用已加入的点集诱导割。两者安全依据相同，但维护的候选不同。

struct DSU {
    vector<int> p,sz;
    explicit DSU(int n):p(n),sz(n,1){ iota(p.begin(),p.end(),0); }
    int find(int x){ return p[x]==x ? x : p[x]=find(p[x]); }
    bool unite(int a,int b){
        a=find(a); b=find(b);
        if(a==b) return false;
        if(sz[a]<sz[b]) swap(a,b);
        p[b]=a; sz[a]+=sz[b]; return true;
    }
};
optional<long long> kruskal(int n, vector<tuple<int,int,int>> edges) {
    // 每条边为 (权重,u,v)，无向图；节点编号 0..n-1。
    if(n<=1) return 0LL;
    sort(edges.begin(),edges.end());
    DSU dsu(n);
    long long sum=0; int used=0;
    for(auto [w,u,v]:edges)
        if(dsu.unite(u,v)){
            sum+=w;
            if(++used==n-1) return sum;
        }
    return nullopt;
}

    long long cost;
    int components;

};
PrimForestResult densePrimForest(const vector<vector<long long>>& weight) {
    const long long INF = LLONG_MAX / 4;
    int n = (int)weight.size();
    if (n > 1000000) throw invalid_argument("size");
    for (const auto& row : weight) if ((int)row.size() != n)
        throw invalid_argument("not square");
    for (int u = 0; u < n; ++u) for (int v = 0; v < n; ++v) {
        long long w = weight[u][v];
        if (w != weight[v][u]) throw invalid_argument("not undirected");
        if (w != INF && (w < -1000000000LL || w > 1000000000LL))
            throw invalid_argument("weight");
    }
    vector<long long> key(n, INF);
    vector<int> parent(n, -1);
    vector<char> used(n, false);
    PrimForestResult result{0,0,{}};
    for (int step = 0; step < n; ++step) {
        int u = -1;
        for (int v = 0; v < n; ++v)
            if (!used[v] && (u == -1 || key[v] < key[u])) u = v;
        if (key[u] == INF) { key[u] = 0; ++result.components; }
        used[u] = true; result.cost += key[u];
        if (parent[u] != -1) result.edges.push_back({parent[u],u});
        for (int v = 0; v < n; ++v)
            if (!used[v] && weight[u][v] < key[v]) {
                key[v] = weight[u][v]; parent[v] = u;
            }
    }
    return result;
}

5.6 Kruskal 的实现细节与稀疏 Prim

Kruskal 把边表按权重从小到大排序，对每条边查询两个端点所在集合，只有不同集合才合并并累计权重。路径压缩与按大小合并使并查集操作均摊接近常数，但排序仍是主要成本。不要在读入边时就按输入顺序合并并累加，否则只是得到某棵生成树，不保证最小。

稀疏图可用堆 Prim：从当前树的边界选最小单边权，弹出后若终点已在树内则跳过；首次接入该点才累计该边权，并把它的出边加入候选。惰性边堆实现 O(E log(E+2))，空间 O(V+E)。这个“已加入”的标记不同于 Dijkstra 在入堆时标记的错误做法。

经典考点与练习

练习：LeetCode 1584 连接所有点的最小费用、684 冗余连接。

面试追问：关键点完整解答

最短路树一定是最小生成树吗？

不一定。AB=AC=2、BC=1，从 A 的最短路树取 AB 与 AC，总成本 4；MST 取一条权 2 的边和 BC，总成本 3，却牺牲 A 到某个点的路径长度。先说清优化目标是所有购买边之和，还是各点到源的距离。

负边会破坏 Kruskal 或 Prim 的正确性吗？

不会。割性质只比较边权大小，不要求非负；树本身没有可重复绕行的环，不存在通过绕负环无限降低的问题。负自环不能加入树，负平行边中也要按是否连接不同分量判断。不能把 Dijkstra 的非负条件搬给 MST。

选了 V-1 条边就一定是生成树吗？

还需要无环且覆盖这 V 个点。Kruskal 每次只合并两个不同分量，因此不会成环，从 V 个分量出发选 V-1 次后恰好剩一个分量，才可确定得到生成树。任意随手选出的 V-1 条边可能含环并遗漏某些点。

跨割的最轻边是否出现在每一棵 MST 中？

若它在该割上严格最轻，则任意 MST 缺少它都会在加边形成的环上有更重跨割边，可交换得到更低总权，矛盾，所以必须包含。若有同权最小边，只保证某个最优解可以包含它，不能断言所有解都包含。

Kruskal、Prim 如何选择？

已有稀疏边表时，Kruskal 直接排序很自然；已有稀疏邻接表时可用堆 Prim；完全图或距离可即时计算时，数组 Prim 的 O(V²) 版本能避免显式建立、排序 O(V²) 条边。选择依据包括输入表示和图密度，不是某算法永远更快。

有向图也能把边拿去 Kruskal 吗？

不能直接套。并查集忽略方向，只判断无向连通，无法保证从指定根沿有向边到达所有节点。对应的有向最小树形图是另一类问题，需专门算法；面试时应先向题目确认连接是否无向。

章末自测：两个综合关键点

怎样验证输出，而不只检查与某一个标准序列相同？

拓扑序不唯一，应验证它包含每个节点一次，且每条边 u→v 满足 pos[u]<pos[v]；返回的环应检查首尾相同且相邻节点确有边。最短路路径应检查端点、边合法和总代价与距离一致。MST 应检查边数、无环、连通和总权，不能仅与某一棵树的边集比较。

为什么测试需要小规模独立参照，而不只跑同一个模板两遍？

两个相同模板容易共享同一种错误。可用 BFS 对照单位权 Dijkstra、Bellman-Ford 对照非负图 Dijkstra、Floyd 对照所有源的 Bellman-Ford、暴力枚举染色对照二分图、小图枚举生成树对照 Prim。随机测试固定种子以便复现，再补不可达负环、平行边、自环、空图等定向用例；测试通过也不能替代理论证明。

bool bruteBipartite(const vector<vector<int>>& g) {
    int n = (int)g.size();
    for (int mask = 0; mask < (1 << n); ++mask) {
        bool ok = true;
        for (int u = 0; u < n; ++u) for (int v : g[u])
            if (((mask >> u) & 1) == ((mask >> v) & 1)) ok = false;
        if (ok) return true;
    }
    return false;
}
optional<long long> bruteTree(int n, const vector<tuple<int,int,int>>& edges) {
    if (n <= 1) return 0LL;
    optional<long long> best;
    int m = (int)edges.size(); // 测试仅 n<=5 的简单无向图，m<=10
    for (int mask = 0; mask < (1 << m); ++mask) {
        int count = 0;
        for (int i = 0; i < m; ++i) count += (mask >> i) & 1;
        if (count != n - 1) continue;
        vector<vector<int>> g(n); long long sum = 0;
        for (int i = 0; i < m; ++i) if ((mask >> i) & 1) {
            auto [w,u,v] = edges[i]; sum += w;
            g[u].push_back(v); g[v].push_back(u);
        }
        vector<char> seen(n); queue<int> q; q.push(0); seen[0] = true;
        int reached = 0;
        while (!q.empty()) {
            int u = q.front(); q.pop(); ++reached;
            for (int v : g[u]) if (!seen[v]) { seen[v] = true; q.push(v); }
        }
        if (reached == n && (!best || sum < *best)) best = sum;
    }
    return best;
}

    }

}
